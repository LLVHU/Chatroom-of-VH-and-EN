<script>
  const joinScreen = document.getElementById('join-screen');
  const chatScreen = document.getElementById('chat-screen');
  const identityBtns = document.querySelectorAll('.identity-btn');
  const messagesEl = document.getElementById('messages');
  const textInput = document.getElementById('text-input');
  const sendBtn = document.getElementById('send-btn');
  const statusEl = document.getElementById('status');

  let ws = null;
  let myName = '';
  let myRole = '';

  function connect() {
    const proto = location.protocol === 'https:' ? 'wss' : 'ws';
    ws = new WebSocket(`${proto}://${location.host}`);

    ws.addEventListener('open', () => {
      statusEl.textContent = 'Online';
      statusEl.className = 'online';
      ws.send(JSON.stringify({ type: 'join', name: myName, role: myRole }));
    });

    ws.addEventListener('close', () => {
      statusEl.textContent = 'Disconnected — reconnecting…';
      statusEl.className = 'offline';
      setTimeout(connect, 1500);
    });

    ws.addEventListener('message', (event) => {
      const data = JSON.parse(event.data);
      if (data.type === 'message') addMessage(data);
      if (data.type === 'system') addSystem(data.text);
    });
  }

  function addMessage(data) {
    const div = document.createElement('div');
    const mine = data.name === myName;
    div.className = 'msg ' + (mine ? 'me' : 'them');
    const time = new Date(data.time).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
    div.innerHTML = `<div class="who">${escapeHtml(data.name)}</div><div>${escapeHtml(data.text)}</div>`;
    messagesEl.appendChild(div);
    messagesEl.scrollTop = messagesEl.scrollHeight;
  }

  function addSystem(text) {
    const div = document.createElement('div');
    div.className = 'system';
    div.textContent = text;
    messagesEl.appendChild(div);
    messagesEl.scrollTop = messagesEl.scrollHeight;
  }

  function escapeHtml(str) {
    const d = document.createElement('div');
    d.textContent = str;
    return d.innerHTML;
  }

  function joinChat(role, name) {
    myName = name;
    myRole = role;
    joinScreen.style.display = 'none';
    chatScreen.style.display = 'flex';
    connect();
    textInput.focus();
  }

  function sendMessage() {
    const text = textInput.value.trim();
    if (!text || !ws || ws.readyState !== WebSocket.OPEN) return;
    ws.send(JSON.stringify({ type: 'message', text }));
    textInput.value = '';
  }

  identityBtns.forEach((btn) => {
    btn.addEventListener('click', () => {
      joinChat(btn.dataset.role, btn.dataset.name);
    });
  });
  sendBtn.addEventListener('click', sendMessage);
  textInput.addEventListener('keydown', (e) => { if (e.key === 'Enter') sendMessage(); });
</script>
